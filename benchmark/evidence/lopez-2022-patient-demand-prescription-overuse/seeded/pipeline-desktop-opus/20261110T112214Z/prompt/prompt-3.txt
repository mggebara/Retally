## Paper
Title: Policy Research Working Paper
Methodology: - **Design / identification:** Field RCT with 2,055 acutely ill patients at 60 public community clinics (CSComs) in Bamako, Mali (Nov–Dec 2016). Vouchers for a free course of ACT tablets (simple-malaria treatment) were randomized *within clinic across days*. Clinics were split into 3 geographic cohorts of 20, and each clinic was randomly assigned one of 20 intervention schedules (Figure 2) over a two-week, 6-day observation window: 2 Control days, 2 Doctor Voucher (DV) days and 2 Patient Voucher (PV) days. Under PV, all acutely ill patients got a voucher at arrival; under DV, providers got the vouchers and handed them out at their discretion. Clinics were reimbursed at retail price for redeemed vouchers, so provider and clinic financial incentives were the same in every arm. A separate Patient Information video treatment and a clinic-level RDT-trust training were cross-randomized; these are analysed elsewhere, with interactions shown in Appendix Tables B3–B4.
- **Estimating equation (used throughout):** y_ict = β0 + βP·PV_ct + βD·DV_ct + γ_t + ε_ict, a linear (OLS/LPM) regression of patient i at clinic c on day t on PV and DV dummies, with Control as the omitted group. γ_t are **date fixed effects**, included because the share of PV/DV/C days varies by date. There are no other controls in the baseline specification. Coefficients are in raw units (percentage-point changes for binary outcomes); nothing is standardised.
- **Standard errors and tests:** Standard errors are **clustered at the clinic level** (60 clusters), the level at which rotation schedules were assigned. Reported alongside are the F-test p-value for H0: βP = βD (the key PV-vs-DV comparison) and the joint F-test p-value for βP = βD = 0. Each outcome row is a separate regression (Table 2 balance format).
- **Outcomes:** These are voucher redemption (from administrative voucher data); antimalarial prescription and purchase, split into simple (ACT) and severe treatment; and costs. Severe treatment is coded when the patient received artemisinin monotherapy or quinine as an injectable, or as tablets together with an ACT. Treatment–illness "match" means a malaria-positive patient purchased treatment, or a negative patient did not.
- **Data sources and sample restrictions:**
  - **In-clinic survey:** every consenting acutely ill patient. This means feeling sick plus at least one of fever, chills, sweating, nausea, vomiting, diarrhoea, poor appetite, headache, cough, weakness, fatigue or reduced consciousness.
  - **Home follow-up the next day:** 1,735 patients randomly selected (up to 6 per clinic-day, with oversampling later), of whom 1,495 (86%) were interviewed.
  - **Home RDT** (CareStart HRP2 Pf): 1,126 patients with a valid result. Analyses of illness–treatment match are restricted to this subsample.
  - The Control group has 628 patients (Table 1 descriptives).
  - **Clinic sampling frame:** CSComs in Bamako, Kati and Kalaban Coro. Clinics that had closed, were more than 15 km from Bamako, or were among the 21 working with an NGO were dropped, leaving 60.
- **Randomization and attrition checks:** Table 2 regresses baseline patient and household characteristics on PV/DV using equation (1). Appendix Table B5 tests selection into the home survey, completion and valid RDT by arm (no significant differences). Appendix Table B6 checks balance on selection and interview; patients who consented to the RDT are sicker and more likely to have bought an antimalarial.
- **Heterogeneity by malaria risk:** Following Cohen, Dupas & Schaner (2015), the RDT result is regressed on observable clinic-survey patient characteristics to get predicted malaria risk (π). Treatment effects are then compared across predicted-risk groups, e.g. the patients with the fewest symptoms, to test model Predictions 1–4. These are: PV > DV voucher use means gatekeeping costs; higher any-treatment in PV means patient-driven demand; higher severe treatment in DV means doctor-driven demand; and whether the marginal patients are low- or high-risk. The text is cut off before the exact predictor set and model are given.
- **No ML components:** There is no train/test split, cross-validation or hyperparameter tuning. Inference uses clustered-SE hypothesis tests on the experimental contrasts.

# Lasso via coordinate descent with an active set:
#   minimise RSS(b) + sum_j pen_j * |b_j|
# ---------------------------------------------------------------------------
cd_lasso <- function(y, X, pen, b, xx, tol = 1e-10) {
  thr <- pen / 2
  r <- as.vector(y - X %*% b)
  scl <- max(sqrt(sum(y^2)), 1e-12)
  active <- which(b != 0)
  for (outer in 1:500) {
    if (length(active) > 0) {
      for (sweep in 1:5000) {
        maxd <- 0
        for (j in active) {
          bj <- b[j]
          xj <- X[, j]
          z <- sum(xj * r) + xx[j] * bj
          if (z > thr[j]) nb <- (z - thr[j]) / xx[j]
          else if (z < -thr[j]) nb <- (z + thr[j]) / xx[j]
          else nb <- 0
          if (nb != bj) {
            d <- nb - bj
            r <- r - xj * d
            b[j] <- nb
            dd <- abs(d) * sqrt(xx[j])
            if (dd > maxd) maxd <- dd
          }
        }
        if (maxd < tol * scl) break
      }
    }
    g <- as.vector(crossprod(X, r))
    viol <- which(b == 0 & abs(g) > thr * (1 + 1e-9))
    if (length(viol) == 0) break
    active <- sort(unique(c(which(b != 0), viol)))
  }
  b
}

# Rigorous (cluster-robust) lasso, as in lassopack's rlasso with cluster()
rlasso_clu <- function(y, X, clust, c0 = 1.1, maxpsiiter = 15, tolpsi = 1e-5) {
  N <- length(y)
  p <- ncol(X)
  clf <- factor(clust)
  G <- nlevels(clf)
  gamma <- 0.1 / log(G)
  lambda <- 2 * c0 * sqrt(N) * qnorm(1 - gamma / (2 * p))
  xx <- colSums(X^2)

  psi_fun <- function(e) {
    S <- rowsum(X * e, clf)
    sqrt(colSums(S^2) / N)
  }
  # initial residuals: OLS on the 5 regressors most correlated with y
  cors <- suppressWarnings(abs(as.vector(cor(X, y))))
  cors[is.na(cors)] <- 0
  top <- order(cors, decreasing = TRUE)[1:min(5, p)]
  e <- lm.fit(X[, top, drop = FALSE], y)$residuals
  psi <- psi_fun(e)
  psi[psi < 1e-12] <- 1e-12

  b <- rep(0, p)
  for (it in 1:maxpsiiter) {
    b <- cd_lasso(y, X, lambda * psi, b, xx)
    sel <- which(b != 0)
    if (length(sel) > 0) {
      e <- lm.fit(X[, sel, drop = FALSE], y)$residuals
    } else {
      e <- y
    }
    psi_new <- psi_fun(e)
    psi_new[psi_new < 1e-12] <- 1e-12
    conv <- max(abs(psi_new - psi)) < tolpsi
    psi <- psi_new
    if (conv) break
  }
  which(b != 0)
}

# Cluster-robust VCE (no small-sample adjustment, as in pdslasso)
clu_vcov <- function(X, e, clust) {
  XtXi <- solve(crossprod(X))
  S <- rowsum(X * e, factor(clust))
  XtXi %*% crossprod(S) %*% XtXi
}

run_pds <- function(yname) {
  y <- df[[yname]]
  pv <- df$patient_voucher
  dv <- df$doctor_voucher
  cl <- df[[cluvar]]
  stopifnot(length(cl) == nrow(df), length(y) == nrow(df))
  D <- as.matrix(df[, dd_vars, drop = FALSE])
  s <- !is.na(y) & !is.na(pv) & !is.na(dv) & !is.na(cl) & stats::complete.cases(D)
  Xall <- as.matrix(df[s, pen_vars, drop = FALSE])
  # Drop any penalised controls with missing values in this sample
  nacols <- colSums(is.na(Xall)) > 0
  if (any(nacols)) {
    cat("  dropping", sum(nacols), "controls with missing values\n")
    Xall <- Xall[, !nacols, drop = FALSE]
  }
  ys <- y[s]; pvs <- pv[s]; dvs <- dv[s]; cls <- cl[s]
  Z <- cbind(1, D[s, , drop = FALSE])
  qZ <- qr(Z)
  yt <- qr.resid(qZ, ys)
  pvt <- qr.resid(qZ, pvs)
  dvt <- qr.resid(qZ, dvs)
  Xt <- qr.resid(qZ, Xall)
  colnames(Xt) <- colnames(Xall)
  keep <- colSums(Xt^2) > 1e-8
  Xt <- Xt[, keep, drop = FALSE]
  # drop exact duplicate columns
  dup <- duplicated(t(round(Xt, 10)))
  Xt <- Xt[, !dup, drop = FALSE]
  N <- length(yt)

  sel_y <- rlasso_clu(yt, Xt, cls)
  sel_p <- rlasso_clu(pvt, Xt, cls)
  sel_d <- rlasso_clu(dvt, Xt, cls)
  sel <- sort(unique(c(sel_y, sel_p, sel_d)))
  cat(sprintf("  %s: N=%d, selected y:%d PV:%d DV:%d union:%d\n", yname, N,
              length(sel_y), length(sel_p), length(sel_d), length(sel)))
  if (length(sel) > 0) cat("   controls:", paste(colnames(Xt)[sel], collapse = " "), "\n")

  W <- cbind(patient_voucher = pvt, doctor_voucher = dvt, Xt[, sel, drop = FALSE])
  qW <- qr(W)
  if (qW$rank < ncol(W)) {
    W <- W[, qW$pivot[1:qW$rank], drop = FALSE]
  }

  list(bP = b["patient_voucher"], bD = b["doctor_voucher"],
       seP = se["patient_voucher"], seD = se["doctor_voucher"], N = N)
}

                  df[, dd_vars, drop = FALSE])
  d <- d[stats::complete.cases(d), ]
  f <- lm(as.formula(paste("y ~ pv + dv +", paste(dd_vars, collapse = "+"))), data = d)
  list(bP = coef(f)["pv"], bD = coef(f)["dv"], seP = NA, seD = NA, N = nrow(d))
}
