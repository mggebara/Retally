is_num_vec <- function(cols, d) {
  if (length(cols) == 0) return(character(0))

dedupe <- function(lst) {
  out <- list()
  for (k in names(lst)) {
    dup <- FALSE
    for (k2 in names(out)) if (identical(lst[[k]], out[[k2]])) { dup <- TRUE; break }
    if (!dup) out[[k]] <- lst[[k]]
  }
  out
}
