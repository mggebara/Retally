Methodology: - Data and unit of analysis: all House public bills (simple/concurrent resolutions excluded) from the 101st (elected 1988) to the 111th Congress (elected 2008), taken from THOMAS and cross-checked against the Congressional Bills Project and the Fowler cosponsorship data. These are linked to biographical data (ICPSR up to the 104th Congress, then the Biographical Directory), Clerk of the House election returns, and Census district characteristics. The analysis is at the bill level (bill j, sponsor i, district k, Congress t), for all members pooled and separately for Democrats and Republicans.
- Dependent variables, both in levels (no logs): (a) the total number of cosponsors on a bill, which measures coalition building; (b) the percent of cosponsors from the opposite party, out of cosponsors plus the sponsor (that is, divided by n+1 so bills with zero cosponsors are kept), which measures bipartisanship. Key regressor: a Female dummy for the sponsor. Headline coefficients are in raw units (number of cosponsors; percentage points), not standardised.
- Baseline OLS (eq. 4, Table 3): Y_ijkt = α + β·Female_i + c_t + Bill_j + Spons_it + Distr_kt + ε. Controls are added one group at a time across columns: (1) Congress (time) fixed effects only; (2) plus bill category dummies (226 minor topics from congressionalbills.org and 33 committees of referral); (3) Congress FE plus individual controls: tenure, rookie dummy, age, committee chair/ranking member dummy, Ivy League dummy, 5 previous-occupation dummies, and total number of bills sponsored; (4) Congress FE plus district controls: 3 macro-region dummies, % black, % urban, % foreign-born, % over 65, log median income, log population density; (5) preferred/full specification with all bill, individual and district controls. District fixed effects were tried but dropped as too imprecise.

The sample is restricted to members elected in mixed-gender (male vs. female) races, and MV is the female candidate's margin of victory, with a flexible polynomial allowed to differ on each side of the zero threshold. It is run pooled and by party, with McCrary density tests at the threshold. The density is balanced overall and among Democrats but shows a clear discontinuity among Republicans, which makes the by-party RD assumptions more stringent. Matching on district/individual characteristics is used as an additional selection correction.

Predicted Republican vote share comes from an OLS of actual Republican share on the district characteristics above (Figure 1). DW-NOMINATE first dimension and Party Unity Score distributions are compared by gender and party. Heterogeneity is examined by seniority, by women's-issue bills, and by cosponsor gender/quality.
- Inference and evaluation: there is no train/test split or cross-validation. fit_ols <- function(df, y, rhs, fe, weighted = FALSE) {
  fml <- as.formula(paste(y, "~", paste(c("sponsor_female", rhs), collapse = " + "), "|", fe))
  if (weighted) {
    m <- feols(fml, data = df, weights = ~wt, cluster = ~group_sponsor,
               fixef.rm = "none", warn = FALSE, notes = FALSE)
  } else {
    m <- feols(fml, data = df, cluster = ~group_sponsor,
               fixef.rm = "none", warn = FALSE, notes = FALSE)
  }
  used <- obs(m)
  list(b = unname(coef(m)["sponsor_female"]),
       se = unname(sqrt(diag(vcov(m)))["sponsor_female"]),
       n = nobs(m),
       nc = length(unique(df$group_sponsor[used])))
}

# probit mimicking Stata: iteratively drop v2 levels and covariates whose nonzero values
# perfectly predict the outcome (dropping those observations and the variable),
# omit collinear covariates, then predict on newdata with the reduced model
pscore_probit <- function(train, newdata, covs) {
  vars <- c("sponsor_female", "v2", covs)
  train <- train[complete.cases(train[, vars, drop = FALSE]), , drop = FALSE]
  dropped_lv <- c()
  dropped_cv <- c()
  repeat {
    changed <- FALSE
    lv <- sort(unique(train$v2))
    perfect <- lv[sapply(lv, function(l) {
      yy <- train$sponsor_female[train$v2 == l]
      all(yy == 0) || all(yy == 1)
    })]
    if (length(perfect) > 0 && length(perfect) < length(lv)) {
      train <- train[!(train$v2 %in% perfect), , drop = FALSE]
      dropped_lv <- c(dropped_lv, perfect)
      changed <- TRUE
    }
    for (cv in covs) {
      x <- as.numeric(train[[cv]])
      nz <- x != 0
      if (any(nz) && any(!nz)) {
        yy <- train$sponsor_female[nz]
        if (all(yy == 0) || all(yy == 1)) {
          train <- train[!nz, , drop = FALSE]
          covs <- setdiff(covs, cv)
          dropped_cv <- c(dropped_cv, cv)
          changed <- TRUE
          break
        }
      }
    }
    if (!changed) break
  }
  keep_lv <- sort(unique(train$v2))
  dum_lv <- if (length(keep_lv) > 1) keep_lv[-1] else numeric(0)
  mk <- function(D) {
    X <- matrix(1, nrow(D), 1)
    colnames(X) <- "cons"
    for (l in dum_lv) {
      X <- cbind(X, as.numeric(D$v2 == l))
      colnames(X)[ncol(X)] <- paste0("v2_", l)
    }
    for (cv in covs) {
      X <- cbind(X, as.numeric(D[[cv]]))
      colnames(X)[ncol(X)] <- cv
    }
    X
  }
  Xt <- mk(train)
  fit <- suppressWarnings(glm(train$sponsor_female ~ Xt - 1, family = binomial(link = "probit")))
  b <- coef(fit)
  b[is.na(b)] <- 0
  Xn <- mk(newdata)
  xb <- as.vector(Xn %*% b)
  p <- pnorm(xb)
  # Stata's predict after probit (default, no 'rules'/'asif') returns missing for
  # observations excluded by the perfect-prediction rules
  excl <- rep(FALSE, nrow(newdata))
  if (length(dropped_lv) > 0) excl <- excl | (!is.na(newdata$v2) & newdata$v2 %in% dropped_lv)
  for (cv in dropped_cv) {
    x <- as.numeric(newdata[[cv]])
    excl <- excl | (!is.na(x) & x != 0)
  }
  p[excl] <- NA
  p
}

make_wt <- function(p, f) {
  w <- ifelse(f == 1, 1 / p, 1 / (1 - p))
  w[!is.finite(w)] <- NA
  w
}

get_band <- function(df, y) {
  s <- df[!is.na(df[[y]]) & !is.na(df$MV1_female), , drop = FALSE]
  h <- tryCatch({
    f <- get("rdbwselect_2014", envir = asNamespace("rdrobust"))
    out <- f(y = s[[y]], x = s$MV1_female, c = 0, kernel = "uniform")
    bw <- out$bws
    if (is.matrix(bw)) bw[1, 1] else as.numeric(bw)[1]
  }, error = function(e) {
    message("rdbwselect_2014 failed: ", conditionMessage(e))
    NA
  })
  if (is.na(h)) {
    h <- tryCatch({
      out <- rdbwselect(y = s[[y]], x = s$MV1_female, c = 0, kernel = "uniform", bwselect = "mserd")
      out$bws[1, 1]
    }, error = function(e) { message("rdbwselect failed: ", conditionMessage(e)); NA })
  }
  h
}
