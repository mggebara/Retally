Block randomization was stratified on (i) geographic location, (ii) treatment status in the 2016 campaign and (iii) the chief's past tax-collection experience. The randomization was run 100 times and the run with the minimum t-statistics on balance checks across eight neighborhood-level variables was kept (Banerjee et al. 2017). The main estimator is OLS comparing Local to Central. Analyses use de facto assignment: one CXL neighborhood received Local by mistake.

The key regressor is the Local treatment dummy, with Central as the omitted category; other arms enter as separate treatment indicators where relevant. This is a linear probability / levels specification with no logs and no standardization. The headline β is in percentage points of compliance, or Congolese francs (CF) of revenue per property.
- **Outcomes and units:** The main outcomes come from administrative data: handheld receipt-printer payment records matched by tax ID to the property register for the universe of registered properties. Secondary outcomes include bribes, assessment accuracy, exemptions, visits, tax morale and trust. These come from the midline survey (36,130 complete; every property, 2–4 weeks after collection) and the endline survey (3,893 baseline respondents re-tracked Mar–Sep 2019; baseline 4,246 households, about 12 per neighborhood).

# ---- treatment code mapping (target coding: 0 control, 1 central, 2 local, 3 CLI, 4 CXL)
map_tmt <- function(v) {
  lab <- NULL
  if (haven::is.labelled(v)) {
    lab <- as.character(haven::as_factor(v, levels = "labels"))
  } else if (is.character(v) || is.factor(v)) {
    lab <- as.character(v)
  }
  if (!is.null(lab)) {
    l <- tolower(lab)
    out <- rep(NA_real_, length(l))
    out[grepl("control|pure|declar", l)] <- 0
    out[is.na(out) & grepl("info|cli", l)] <- 3
    out[is.na(out) & grepl("cxl| x |cross|both|\\*", l)] <- 4
    out[is.na(out) & grepl("local|chief|chef", l)] <- 2
    out[is.na(out) & grepl("central|state|dgrkoc", l)] <- 1
    ok <- !is.na(lab) & lab != ""
    if (sum(ok) > 0 && mean(!is.na(out[ok])) > 0.9) return(out)
  }
  num(v)
}

# ---- Stata daily date conversion
to_td <- function(x) {
  tryCatch({
    if (inherits(x, "Date")) {
      as.numeric(x) + 3653
    } else if (inherits(x, "POSIXt")) {
      as.numeric(as.Date(x, tz = "UTC")) + 3653
    } else if (is.numeric(x) || haven::is.labelled(x)) {
      v <- num(x)
      m <- suppressWarnings(median(v, na.rm = TRUE))
      if (is.na(m)) rep(NA_real_, length(v))
      else if (m > 20000 && m < 23000) v
      else if (m > 1.7e12 && m < 2.0e12) floor(v / 86400000)
      else if (m > 17000 && m < 19000) v + 3653
      else rep(NA_real_, length(v))
    } else if (is.character(x)) {
      p <- suppressWarnings(lubridate::parse_date_time(x,
        orders = c("Ymd", "dmY", "mdY", "bdY", "dbY", "Ymd HMS", "bdY HMS", "mdY HMS", "dmY HMS", "Ymd HM"),
        quiet = TRUE))
      as.numeric(as.Date(p)) + 3653
    } else rep(NA_real_, length(x))
  }, error = function(e) rep(NA_real_, length(x)))
}
