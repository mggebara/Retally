safe_relevel <- function(x, patterns) {
  x <- factor(x)
  for (p in patterns) {
    hits <- grep(p, levels(x), ignore.case = TRUE, value = TRUE)
    if (length(hits) > 0) {
      return(relevel(x, ref = hits[1]))
    }
  }
  x
}
