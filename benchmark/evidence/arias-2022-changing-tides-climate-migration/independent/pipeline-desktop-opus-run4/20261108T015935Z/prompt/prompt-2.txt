Title: Changing Tides: Public Attitudes on Climate Migration∗
Methodology: - **Design / data (Study 1):** Choice-based (forced-choice, paired-profile) conjoint experiment fielded via Dynata on nationally representative online samples: 1,086 U.S. and 1,074 German respondents (2,160 total), 18,966 (U.S.) and 18,862 (Germany) profile observations ("unique choice tasks"). Each respondent saw 9 pairs of randomized migrant profiles; after each pair they rated each profile and chose which migrant to admit for settlement in their state. Pre-registered at EGAP (Design ID 20190905AA). U.S. and Germany are analysed separately (no pooling).
- **Attributes and baselines (Table 1):** Seven fully randomized attributes; baseline levels in parentheses: Reason for Migration (Economic Opportunity; vs Drought, Flooding, Wildfires, Political/Religious/Ethnic Persecution), Language Fluency (None; vs Broken, Fluent), Gender (Female; vs Male), Occupation (Unemployed; vs Cleaner, Teacher, Doctor), Religion (Atheist/Agnostic; vs Christian, Muslim; "agnostic" replaced by "atheist" in Germany), Origin (Another region in your country; vs Afghanistan, Ethiopia, Myanmar, Ukraine), Vulnerability (None; vs PTSD, Food insecurity, No surviving family members, Physically handicapped).
- **Estimator / specification:** Average Marginal Component Effects (AMCEs; Hainmueller, Hopkins & Yamamoto 2014), i.e. a linear probability model (OLS) of the binary choice outcome (profile chosen = 1) on dummy indicators for all non-baseline levels of the seven attributes, estimated separately by country. Outcome is a probability; coefficients are unstandardised changes in the probability a profile is preferred relative to the baseline level (e.g., U.S. drought 0.035, flooding 0.037, wildfires 0.040, persecution 0.076; Germany 0.081, 0.086, 0.060, 0.162). - **Standard errors / inference:** Robust standard errors clustered by respondent; Figure 1 shows 90% confidence intervals; Table 2 significance stars at *** p<0.01, ** p<0.05, * p<0.1.
- **Samples / subsets:** Headline (Table 2, Figure 1 left panel) uses all profile pairs ("full model"). Figure 1 middle panel restricts to internal-migrant (within-country origin) profile pairs; right panel to international-migrant (foreign origin) profile pairs. Robustness also excludes internal migrant profiles that may seem implausible.
- **Supplementary analyses:** Marginal means (P(preferred) for economic migrant 0.45-0.48, climate migrant 0.48-0.53, refugee 0.53-0.60), conjoint diagnostic tests, heterogeneous effects by subgroup (Leeper, Hobolt & Tilley 2020; e.g., employed / college-educated respondents), and origin x reason-for-migration interactions; Wald-type comparisons among climatic drivers (no significant within-sample differences between drought, flooding, wildfires).
- **Study 2:** Between-subjects factorial survey experiment on representative U.S. and German samples; after the same sociodemographic pre-test (age, gender, education, religiosity, trust, political interest, ideology, partisanship, foreign policy orientation, empathy), respondents read a mock article priming migration, climate change, or climate migration (domestic or worldwide), with a soccer article as control. DVs are multiple-item importance indices of support for policies to address each issue (mitigation, aiding climate migrants); treatment effects estimated relative to control, with empathy as a correlate and subgroup analyses (details in appendix). Finding: null effects of primes; empathy robustly correlated with support.

read_qualtrics <- function(path) {
  raw <- suppressWarnings(readr::read_csv(path, col_types = readr::cols(.default = readr::col_character()),
                                          progress = FALSE, name_repair = "minimal"))
  nm <- names(raw)
  nm <- sub("^﻿", "", nm)
  names(raw) <- nm
  qtext <- rep(NA_character_, length(nm))
  drop <- integer(0)
  for (r in 1:min(3, nrow(raw))) {
    v <- raw[[ifelse("StartDate" %in% nm, "StartDate", nm[1])]][r]
    v2 <- paste(unlist(raw[r, ]), collapse = " ")
    if (!is.na(v) && (grepl("Start", v) || grepl("ImportId", v) || grepl("ImportId", v2))) {
      if (!grepl("ImportId", v2) && all(is.na(qtext))) qtext <- as.character(unlist(raw[r, ]))
      drop <- c(drop, r)
    }
  }
  if (length(drop) > 0) raw <- raw[-drop, , drop = FALSE]
  if (all(is.na(qtext))) qtext <- nm
  list(data = raw, qtext = qtext)
}

estimate <- function(path) {
  b <- build_long(path)
  df <- b$df
  attrs <- b$attrs
  # identify reason attribute
  reason_attr <- NULL
  for (a in attrs) {
    lv <- unique(df[[a]])
    if (any(grepl(pat_drought, lv)) && any(grepl(pat_pers, lv))) { reason_attr <- a; break }
  }
  if (is.null(reason_attr)) {
    for (a in attrs) if (any(grepl(pat_drought, unique(df[[a]])))) { reason_attr <- a; break }
  }
  if (is.null(reason_attr)) stop("Reason attribute not found")
  rhs <- c()
  reason_levels <- NULL
  for (ai in seq_along(attrs)) {
    a <- attrs[ai]
    lv <- sort(unique(df[[a]]))
    base <- lv[1]
    if (a == reason_attr) {
      eb <- lv[grepl(pat_econ, lv)]
      if (length(eb) > 0) base <- eb[1]
    } else {
      cand <- lv[grepl("(?i)^none$|^keine?$|unemployed|arbeitslos|^female$|weiblich|^frau$|atheist|agnostic|konfessionslos|another region|anderen region|andere region|region in", lv)]
      if (length(cand) > 0) base <- cand[1]
    }
    ord <- c(base, setdiff(lv, base))
    vname <- paste0("A", ai)
    df[[vname]] <- factor(match(df[[a]], ord), levels = seq_along(ord), labels = paste0("L", seq_along(ord)))
    rhs <- c(rhs, vname)
    if (a == reason_attr) { reason_levels <- ord; reason_var <- vname }
  }
  fml <- as.formula(paste("chosen ~", paste(rhs, collapse = " + ")))
  m <- feols(fml, data = df, cluster = ~rid)
  cf <- coef(m)
  se <- sqrt(diag(vcov(m)))
  get_lv <- function(pat) {
    i <- which(grepl(pat, reason_levels))
    i <- setdiff(i, 1)
    if (length(i) == 0) return(c(NA, NA))
    cn <- paste0(reason_var, "L", i[1])
    c(unname(cf[cn]), unname(se[cn]))
  }
  # wildfire pattern should not hit other levels; check order of matching
  out <- list(
    drought = get_lv(pat_drought),
    flooding = get_lv(pat_flood),
    wildfires = get_lv(pat_fire),
    persecution = get_lv(pat_pers),
    n_resp = length(unique(df$rid)),
    n_obs = nobs(m)
  )
  print(summary(m))
  message("Reason levels: ", paste(reason_levels, collapse = " | "))
  out
}
