## Paper
Title: Changing Tides: Public Attitudes on Climate Migration∗
Methodology: **Methodology Summary**

- **Design — Study 1 (Conjoint Experiment):** Choice-based conjoint embedded in nationally representative online surveys (Dynata) of 1,086 U.S. and 1,074 German respondents, yielding 18,966 and 18,862 choice tasks. Each respondent evaluated 9 pairs of randomized migrant profiles varying on 7 attributes (reason for migration, language fluency, gender, occupation, religion, origin, vulnerability). Pre-test battery collected sociodemographics, ideology, partisanship, empathy, etc. Pre-registered with EGAP (Design ID: 20190905AA).

- **Focal attribute & levels:** Reason for migration took 5 levels (economic opportunity [baseline], drought, flooding, wildfires, political/religious/ethnic persecution). Origin levels included one internal ("another region in your country") and four international origins (Afghanistan, Ethiopia, Myanmar, Ukraine), enabling internal vs. international subset analyses.

- **Estimation — Study 1:** Average Marginal Component Effects (AMCEs) following Hainmueller, Hopkins, and Yamamoto (2014). Dependent variable: binary indicator that a profile was chosen for settlement in the respondent's state. Headline AMCEs are in percentage-point (probability) units, not standardized. Standard errors are robust and clustered by respondent. Robustness via marginal means, heterogeneous treatment effects (Leeper, Hobolt, Tilley 2020), subset analyses (internal vs. international, excluding implausible internal profiles), and interactions of origin × reason for migration.

- **Headline AMCEs (Table 2, vs. economic-opportunity baseline):** U.S.: drought +3.5pp, flooding +3.7pp, wildfires +4.0pp, persecution +7.6pp. Germany: drought +8.1pp, flooding +8.6pp, wildfires +6.0pp, persecution +16.2pp.

- **Design — Study 2 (Factorial Experiment):** Between-subjects factorial on representative U.S./German samples. After the pre-test, respondents saw a mock article priming migration, climate change, or climate migration (country-level or worldwide); control group received a soccer article. Outcomes were multi-item indices measuring support for policies to address migration, climate change, and climate migration.

- **Evaluation — Study 2:** Tests of whether priming climate migration/climate change shifts policy support indices; subgroup analyses to test consistency of null effects; empathy examined as a correlate of mitigation and pro-migrant policy support.

- **Key hyperparameter/design choices:** 9 conjoint tasks per respondent; attributes drawn from Spilker et al. (2020) and Bansak et al. (2016) plus wildfires added; "atheist" substituted for "agnostic" in Germany for cultural relevance.

build_conjoint_long <- function(df, country) {
  cn <- names(df)
  
  hdr_cols <- cn[grepl("^F-[0-9]+-[0-9]+$", cn)]
  if (length(hdr_cols) == 0) hdr_cols <- cn[grepl("^F\\.[0-9]+\\.[0-9]+$", cn)]
  
  val_cols <- cn[grepl("^F-[0-9]+-[0-9]+-[0-9]+$", cn)]
  if (length(val_cols) == 0) val_cols <- cn[grepl("^F\\.[0-9]+\\.[0-9]+\\.[0-9]+$", cn)]
  
  sep <- if (any(grepl("^F-", cn))) "-" else "."
  pat <- paste0("^F\\", sep)
  
  hdr_parts <- do.call(rbind, strsplit(gsub(pat, "", hdr_cols), paste0("[", sep, "]")))
  val_parts <- do.call(rbind, strsplit(gsub(pat, "", val_cols), paste0("[", sep, "]")))
  
  hdr_tasks <- as.integer(hdr_parts[,1])
  hdr_slots <- as.integer(hdr_parts[,2])
  
  val_tasks  <- as.integer(val_parts[,1])
  val_profs  <- as.integer(val_parts[,2])
  val_slots  <- as.integer(val_parts[,3])
  
  n_tasks <- max(c(hdr_tasks, val_tasks))
  n_attrs <- max(c(hdr_slots, val_slots))
  n_profs <- max(val_profs)
  
  choice_candidates <- c()
  for (col in cn) {
    if (col %in% hdr_cols || col %in% val_cols) next
    v <- df[[col]]
    v_num <- suppressWarnings(as.numeric(v))
    nn <- sum(!is.na(v_num))
    if (nn > nrow(df) * 0.3) {
      uv <- unique(v_num[!is.na(v_num)])
      if (length(uv) <= 3 && length(uv) >= 1 && all(uv %in% c(1,2))) {
        choice_candidates <- c(choice_candidates, col)
      }
    }
  }
  compl <- sapply(choice_candidates, function(cc) sum(!is.na(suppressWarnings(as.numeric(df[[cc]])))))
  choice_candidates <- choice_candidates[order(-compl)]
  choice_cols <- head(choice_candidates, n_tasks)
  choice_cols <- choice_cols[order(match(choice_cols, cn))]
  
  rows <- list()
  resp_ids <- df$ResponseId
  
  for (t in 1:n_tasks) {
    chc <- suppressWarnings(as.numeric(df[[choice_cols[t]]]))
    for (p in 1:n_profs) {
      prof_data <- data.frame(ResponseId = resp_ids, task = t, profile = p,
                              chosen = as.integer(chc == p),
                              stringsAsFactors = FALSE)
      for (s in 1:n_attrs) {
        vcol <- sprintf("F%s%d%s%d%s%d", sep, t, sep, p, sep, s)
        hcol <- sprintf("F%s%d%s%d", sep, t, sep, s)
        if (vcol %in% cn) {
          prof_data[[paste0("slot", s, "_value")]] <- df[[vcol]]
        }
        if (hcol %in% cn) {
          prof_data[[paste0("slot", s, "_name")]] <- df[[hcol]]
        }
      }
      rows[[length(rows)+1]] <- prof_data
    }
  }
  
  long <- do.call(rbind, rows)
  long$country <- country
  attr(long, "n_attrs") <- n_attrs
  long
}

extract_attr_by_name <- function(long, name_patterns) {
  n_attrs <- attr(long, "n_attrs")
  out <- rep(NA_character_, nrow(long))
  for (s in 1:n_attrs) {
    name_col <- paste0("slot", s, "_name")
    val_col  <- paste0("slot", s, "_value")
    if (!(name_col %in% names(long)) || !(val_col %in% names(long))) next
    nm <- tolower(as.character(long[[name_col]]))
    match_rows <- rep(FALSE, length(nm))
    for (pat in name_patterns) {
      match_rows <- match_rows | grepl(pat, nm, ignore.case = TRUE)
    }
    match_rows[is.na(match_rows)] <- FALSE
    out[match_rows] <- as.character(long[[val_col]])[match_rows]
  }
  out
}

standardize_reason <- function(x) {

  r[grepl("econom|wirtschaft|ökonom|oekonom|arbeit|job", x)] <- "economic"

prep_for_amce <- function(long) {
  long <- long[!is.na(long$chosen) & !is.na(long$reason_std), ]
  long$reason_std <- factor(long$reason_std, levels = c("economic","drought","flooding","wildfires","persecution"))
  for (ac in c("vulnerability","language","gender","origin","religion","occupation")) {
    if (ac %in% names(long)) long[[ac]] <- factor(long[[ac]])
  }
  long
}

build_formula <- function(long) {
  other_cols <- intersect(c("vulnerability","language","gender","origin","religion","occupation"), names(long))
  keep <- other_cols[sapply(other_cols, function(c) {
    v <- long[[c]]
    v <- v[!is.na(v)]
    length(unique(v)) > 1
  })]
  rhs <- paste(c("reason_std", keep), collapse = " + ")
  as.formula(paste("chosen ~", rhs))
}

run_mod <- function(long) {
  if (nrow(long) == 0) return(NULL)
  fml <- build_formula(long)
  tryCatch(feols(fml, data = long, cluster = ~ResponseId),
           error = function(e) {
             # fallback: just reason
             tryCatch(feols(chosen ~ reason_std, data = long, cluster = ~ResponseId),
                      error = function(e2) NULL)
           })
}

us_mod <- run_mod(us_long_m)
de_mod <- run_mod(de_long_m)

get_coef <- function(mod, name) {
  if (is.null(mod)) return(NA_real_)
  co <- coef(mod)
  v <- co[name]
  if (is.na(v)) {
    matches <- grep(name, names(co), value = TRUE, fixed = TRUE)
    if (length(matches) > 0) v <- co[matches[1]]
  }
  as.numeric(v)
}
