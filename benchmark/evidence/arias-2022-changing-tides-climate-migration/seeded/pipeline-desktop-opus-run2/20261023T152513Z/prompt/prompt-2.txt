## Paper
Title: Changing Tides: Public Attitudes on Climate Migration∗
Methodology: - **Design (Study 1):** Choice-based (paired-profile, forced-choice) conjoint experiment fielded with Dynata on nationally representative online samples: 1086 U.S. and 1074 German respondents (2160 in total). Each respondent saw 9 pairs of randomized migrant profiles, giving 18,966 (U.S.) and 18,862 (Germany) profile observations ("unique choice tasks"). Seven attributes were fully randomized (Table 1), each with a baseline level: Reason for Migration (baseline Economic Opportunity; Drought, Flooding, Wildfires, Political/Religious/Ethnic Persecution), Language Fluency (baseline None; Broken, Fluent), Gender (baseline Female; Male), Occupation (baseline Unemployed; Cleaner, Teacher, Doctor), Religion (baseline Atheist/Agnostic; Christian, Muslim), Origin (baseline Another region in your country; Afghanistan, Ethiopia, Myanmar, Ukraine), and Vulnerability (baseline None; PTSD, Food insecurity, No surviving family members, Physically handicapped). "Agnostic" was replaced by "atheist" in Germany. A pre-treatment sociodemographic battery measured age, gender, education, religiosity, trust in government, political interest, ideology, partisanship, foreign policy orientation and empathy.
- **Estimand / estimating equation:** Average Marginal Component Effects (AMCEs; Hainmueller, Hopkins and Yamamoto 2014). These come from a linear probability model (OLS) estimated separately for each country. The dependent variable is a binary indicator that the profile was the one chosen for settlement in the respondent's state. The regressors are dummy indicators for every non-baseline level of all seven attributes, entered together in one additive model. Coefficients are unstandardized changes in the probability (0–1 scale; reported in the text as percentage points) that a profile is preferred, relative to the baseline level. There are no fixed effects and no respondent-level controls. Hypotheses (pre-registered at EGAP): H1, climate reasons > economic opportunity; H2, climate reasons < persecution.
- **Standard errors / inference:** Robust standard errors clustered by respondent. Table 2 stars are *** p<0.01, ** p<0.05, * p<0.1. Figure 1 shows 90% confidence intervals.
- **Headline results (Table 2, full model / left panel of Fig. 1):** U.S.: Drought 0.035 (0.012), Flooding 0.037 (0.012), Wildfires 0.040 (0.011), Persecution 0.076 (0.012). Germany: Drought 0.081 (0.012), Flooding 0.086 (0.012), Wildfires 0.060 (0.012), Persecution 0.162 (0.013). Table 2 also reports AMCEs for all other attributes (for example, U.S. Doctor 0.187, Fluent 0.104, Male −0.044, Muslim −0.047).
- **Sample splits / subsets:** The full model uses all profile pairs. Separate models restrict to internal-migrant (within-country origin) profile pairs (Fig. 1, middle panel) and to international-migrant (between-country) profile pairs (Fig. 1, right panel). Robustness checks in the appendix exclude internal migrant profiles that may seem implausible.
- **Additional analyses:** Marginal means are reported as the probability a profile is preferred: economic migrant 0.45–0.48, climate migrants 0.48–0.53, refugees 0.53–0.60. The appendix also covers conjoint diagnostic tests, heterogeneity via subgroup marginal means (Leeper, Hobolt and Tilley 2020), origin × reason interactions, and heterogeneity by respondent employment and college education.
- **Study 2:** Between-subjects factorial experiment on representative U.S. and German samples. After the same pre-test, respondents read a mock article priming migration, climate change, or climate migration, framed for the respondent's country or worldwide. The control group read an article about soccer. Outcomes are multi-item importance indices of support for policies addressing each issue (climate mitigation, aiding climate migrants). Treatment effects are estimated against the control group, with empathy examined as a correlate. Results: no significant priming effects, while empathy is robustly correlated with support.
- **No ML components:** There is no train/test split, cross-validation or hyperparameter tuning. Evaluation is by the sign, size and significance of AMCEs relative to the baseline (economic opportunity).

read_raw <- function(path, enc = "UTF-8") {
  raw <- readr::read_csv(path, col_types = readr::cols(.default = readr::col_character()),
                         na = character(), progress = FALSE, name_repair = "minimal",
                         locale = readr::locale(encoding = enc))
  raw <- as.data.frame(raw, stringsAsFactors = FALSE, optional = TRUE)
  nm <- sub("^﻿", "", names(raw))
  nm <- gsub("^\"|\"$", "", nm)
  names(raw) <- nm
  raw
}

code_choice <- function(mat) {
  v <- trimws(as.vector(mat)); v[is.na(v)] <- ""
  nonempty <- v[v != ""]
  num <- suppressWarnings(as.numeric(gsub("[^0-9]", "", nonempty)))
  if (length(nonempty) > 0 && all(!is.na(num)) && all(num %in% c(1, 2))) {
    n2 <- suppressWarnings(as.numeric(gsub("[^0-9]", "", v)))
    out <- ifelse(v == "", NA, n2)
  } else {
    tab <- sort(table(nonempty), decreasing = TRUE)
    top <- names(tab)[1:min(2, length(tab))]
    tn <- suppressWarnings(as.numeric(gsub("[^0-9]", "", top)))
    top <- if (all(!is.na(tn))) top[order(tn)] else sort(top)
    out <- match(v, top)
  }
  matrix(out, nrow = nrow(mat))
}

read_conjoint <- function(path, responses) {
  raw <- read_raw(path, "UTF-8")
  nm <- names(raw)
  lv_cols <- grep("^[A-Za-z]+-[0-9]+-[0-9]+-[0-9]+$", nm, value = TRUE)
  if (length(lv_cols) == 0) stop("No conjoint level columns found in ", path)
  # encoding check
  chk <- unlist(raw[, lv_cols[seq_len(min(length(lv_cols), 200))], drop = FALSE])
  if (any(!validUTF8(chk))) {
    raw <- read_raw(path, "latin1")
    nm <- names(raw)
  }
  idcol <- if ("ResponseId" %in% nm) "ResponseId" else nm[gl("responsei", nm)][1]
  ids <- raw[[idcol]]
  hdr <- ids %in% c("Response ID", "ResponseID") | grepl("ImportId", ids, fixed = TRUE)
  raw <- raw[!hdr, , drop = FALSE]
  nr <- nrow(raw)

  letter <- names(sort(table(sub("-.*$", "", lv_cols)), decreasing = TRUE))[1]
  a_cols <- grep(paste0("^", letter, "-1-[0-9]+$"), nm, value = TRUE)
  nattr <- max(as.integer(sub(".*-", "", a_cols)))

  ch <- code_choice(sapply(responses, function(cn) if (cn %in% names(raw)) raw[[cn]] else rep("", nr)))

  long <- list()
  for (k in seq_along(responses)) for (p in 1:2) for (j in seq_len(nattr)) {
    an <- raw[[sprintf("%s-%d-%d", letter, k, j)]]
    lv <- raw[[sprintf("%s-%d-%d-%d", letter, k, p, j)]]
    if (is.null(an) || is.null(lv)) next
    long[[length(long) + 1]] <- data.frame(r = seq_len(nr), task = k, profile = p,
                                           aname = trimws(an), lvl = trimws(lv),
                                           stringsAsFactors = FALSE)
  }
  long <- bind_rows(long)
  long <- long[!is.na(long$aname) & long$aname != "" & !is.na(long$lvl) & long$lvl != "", ]
  anames <- unique(long$aname)
  canon <- vapply(anames, function(a) classify_attr(unique(long$lvl[long$aname == a]), a), character(1))
  long$attr <- unname(canon[long$aname])
  long <- long[!is.na(long$attr), ]

  wide <- long %>%
    dplyr::select(r, task, profile, attr, lvl) %>%
    tidyr::pivot_wider(id_cols = c(r, task, profile), names_from = attr, values_from = lvl,
                       values_fn = function(x) x[1]) %>%
    as.data.frame(stringsAsFactors = FALSE)
  wide$selected <- ch[cbind(wide$r, wide$task)]
  wide$resp_id <- raw[[idcol]][wide$r]
  wide <- wide[order(wide$r, wide$task, wide$profile), ]
  rownames(wide) <- NULL
  wide
}

# ---------------------------------------------------------------------------
# AMCE estimation (LPM, respondent-clustered SEs). Attribute pairs with
# structural zeros (design restrictions) are modelled with full interaction
# cells and AMCEs averaged over supported levels, as in cjoint::amce.
# ---------------------------------------------------------------------------
estimate_amce <- function(d) {
  bases <- get_bases(d)
  grp <- setNames(seq_along(attrs), attrs)
  for (i in 1:(length(attrs) - 1)) for (j in (i + 1):length(attrs)) {
    tb <- table(d[[attrs[i]]], d[[attrs[j]]])
    ex <- outer(rowSums(tb), colSums(tb)) / sum(tb)
    if (any(tb == 0 & ex > 30)) {
      gi <- grp[attrs[i]]; gj <- grp[attrs[j]]
      grp[grp == gj] <- gi
      cat("Dependency detected:", attrs[i], "x", attrs[j], "\n")
    }
  }
  X <- list(); info <- list(); groups <- list()
  for (g in unique(grp)) {
    members <- names(grp)[grp == g]
    if (length(members) == 1) {
      a <- members
      for (l in setdiff(sort(unique(d[[a]])), bases[[a]])) {
        nm <- paste0("x", length(X) + 1)
        X[[nm]] <- as.numeric(d[[a]] == l)
        info[[nm]] <- list(type = "main", attr = a, level = l)
      }
    } else {
      cell <- do.call(paste, c(lapply(members, function(m) d[[m]]), sep = "||"))
      cells <- sort(unique(cell))
      refcell <- paste(unlist(bases[members]), collapse = "||")
      if (!refcell %in% cells) refcell <- cells[1]
      cellnm <- setNames(rep(NA_character_, length(cells)), cells)
      for (cc in setdiff(cells, refcell)) {
        nm <- paste0("x", length(X) + 1)
        X[[nm]] <- as.numeric(cell == cc)
        cellnm[cc] <- nm
      }
      groups[[length(groups) + 1]] <- list(members = members, cells = cells, cellnm = cellnm)
    }
  }
  dd <- as.data.frame(X)
  dd$selected <- d$selected
  dd$resp_id <- d$resp_id
  fml <- as.formula(paste("selected ~", paste(names(X), collapse = " + ")))
  fit <- fixest::feols(fml, data = dd, cluster = ~resp_id)
  b <- coef(fit); V <- vcov(fit)

  amce <- function(a, l) {
    w <- setNames(rep(0, length(b)), names(b))
    gidx <- which(vapply(groups, function(g) a %in% g$members, logical(1)))
    if (length(gidx) == 0) {
      nm <- names(info)[vapply(info, function(z) z$attr == a && z$level == l, logical(1))]
      if (length(nm) == 0 || !(nm %in% names(b))) return(c(NA, NA))
      w[nm] <- 1
    } else {
      g <- groups[[gidx]]
      parts <- do.call(rbind, strsplit(g$cells, "||", fixed = TRUE))
      colnames(parts) <- g$members
      others <- setdiff(g$members, a)
      okey <- apply(parts[, others, drop = FALSE], 1, paste, collapse = "||")
      combos <- unique(okey)
      sup <- 0
      for (cmb in combos) {
        cl <- g$cells[parts[, a] == l & okey == cmb]
        cb <- g$cells[parts[, a] == bases[[a]] & okey == cmb]
        if (length(cl) == 1 && length(cb) == 1) {
          sup <- sup + 1
          for (pair in list(list(cl, 1), list(cb, -1))) {
            nm <- g$cellnm[pair[[1]]]
            if (!is.na(nm) && nm %in% names(b)) w[nm] <- w[nm] + pair[[2]]
          }
        }
      }
      if (sup == 0) return(c(NA, NA))
      w <- w / sup
    }
    est <- sum(w * b)
    se <- sqrt(as.numeric(t(w) %*% V %*% w))
    c(est, se)
  }

  rl <- setdiff(sort(unique(d$Reason)), bases$Reason)
  keys <- vapply(rl, reason_key, character(1))
  if (sum(is.na(keys)) == 1 && !("flooding" %in% keys)) keys[is.na(keys)] <- "flooding"
  res <- list()
  for (i in seq_along(rl)) {
    if (is.na(keys[i])) next
    res[[keys[i]]] <- amce("Reason", rl[i])
  }
  cat("Baselines:\n"); print(unlist(bases))
  cat("Reason levels:\n"); print(setNames(keys, rl))
  res
}
