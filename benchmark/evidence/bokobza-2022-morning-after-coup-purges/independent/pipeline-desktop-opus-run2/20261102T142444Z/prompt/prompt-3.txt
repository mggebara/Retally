panel_lag <- function(id, t, x) {
  o <- order(id, t)
  idv <- id[o]; tv <- t[o]; xv <- x[o]
  n <- length(xv)
  prev <- c(NA, xv[-n])
  ok <- c(FALSE, !is.na(idv[-1]) & !is.na(idv[-n]) & idv[-1] == idv[-n] &
            !is.na(tv[-1]) & !is.na(tv[-n]) & tv[-1] == tv[-n] + 1)
  prev[!ok] <- NA
  res <- rep(NA_real_, n)
  res[o] <- prev
  res
}
