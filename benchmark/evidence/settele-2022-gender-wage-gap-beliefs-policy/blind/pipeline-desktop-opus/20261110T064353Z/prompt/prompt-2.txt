findv <- function(df, exact = character(0), regex = character(0), labregex = character(0), exclude = EXCL) {

  ok <- !grepl(exclude, nm, ignore.case = TRUE)

likert <- function(x) {
  if (is.null(x)) return(NULL)
  v <- tonum(x)
  lb <- attr(x, "labels")
  if (!is.null(lb) && length(lb) > 1) {
    dk <- lb[grepl("know|prefer|no answer|n/a|not sure", names(lb), ignore.case = TRUE)]
    if (length(dk)) v[v %in% as.numeric(dk)] <- NA
    lb2 <- lb[!lb %in% dk]
    if (length(lb2) > 1) {
      lo <- names(lb2)[which.min(lb2)]
      if (grepl("strongly agree|^agree|very large|large|definitely|strongly support|much more|a lot", lo, ignore.case = TRUE) &&
          !grepl("disagree|small|less|oppose|not ", lo, ignore.case = TRUE)) {
        v <- max(as.numeric(lb2)) + min(as.numeric(lb2)) - v
      }
    }
  }
  v[!is.na(v) & (v >= 90 | v < 0)] <- NA
  v
}

mkfemale <- function(x) {
  if (is.null(x)) return(NULL)
  lb <- attr(x, "labels")
  if (!is.null(lb)) {

  }
  if (is.character(x)) { s <- tolower(trimws(x)); return(ifelse(s == "" | is.na(s), NA, as.numeric(s %in% c("f", "female", "woman")))) }

  if (all(u %in% c(0, 1))) return(v)
  if (all(u %in% c(1, 2))) return(as.numeric(v == 2))
  v
}

dumlab <- function(x, pat, notpat = NULL) {

  r
}

zc <- function(x, ref) {
  m <- mean(x[ref], na.rm = TRUE); s <- sd(x[ref], na.rm = TRUE)
  (x - m) / s
}

anderson <- function(Z, ref) {
  Z <- as.matrix(Z)
  keep <- apply(Z, 2, function(c) sum(!is.na(c)) > 0)
  Z <- Z[, keep, drop = FALSE]
  Zr <- Z[ref, , drop = FALSE]; Zr <- Zr[complete.cases(Zr), , drop = FALSE]
  w <- tryCatch(rowSums(solve(cov(Zr))), error = function(e) rep(1, ncol(Z)))
  cat("  Anderson weights:", round(w, 3), "\n")
  idx <- apply(Z, 1, function(z) { ok <- !is.na(z); if (!any(ok)) NA else sum(w[ok] * z[ok]) / sum(w[ok]) })
  zc(idx, ref)
}
