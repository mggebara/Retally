strip_s4 <- function(x) {
  if (!isS4(x)) return(x)
  vl <- attr(x, 'value.labels', exact = TRUE)
  y <- tryCatch(asS4(x, FALSE, FALSE), error = function(e) NULL)
  if (is.null(y)) return(rep(NA, length(x)))
  attributes(y) <- NULL
  lab <- NULL
  if (!is.null(vl)) {
    lab <- tryCatch({
      vals <- attr(vl, 'values', exact = TRUE)
      if (isS4(vals)) { vals <- asS4(vals, FALSE, FALSE); attributes(vals) <- NULL }
      l <- if (isS4(vl)) asS4(vl, FALSE, FALSE) else vl
      attributes(l) <- NULL
      if (length(vals) == length(l) && length(l) > 0) setNames(as.character(l), as.character(vals)) else NULL
    }, error = function(e) NULL)
  }
  if (!is.null(lab)) attr(y, 'vlab') <- lab
  y
}
