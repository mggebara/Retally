## clean_number function 1 
function (vec, high_numbers = "") 
{
    numeric_vec <- as.numeric(gsub(",", ".", gsub("[[:alpha:]  !#$%&')?/(@:;€_-]", 
        "", vec)))
    if (high_numbers == "remove") {
        is.na(numeric_vec) <- numeric_vec > 10000
    }
    else if (high_numbers == "divide") {
        numeric_vec[numeric_vec > 10000 & !is.na(numeric_vec)] <- numeric_vec[numeric_vec > 
            10000 & !is.na(numeric_vec)]/12
    }
    else if (high_numbers == "divide&remove") {
        numeric_vec[numeric_vec > 10000 & !is.na(numeric_vec)] <- numeric_vec[numeric_vec > 
            10000 & !is.na(numeric_vec)]/12
        is.na(numeric_vec) <- numeric_vec > 6000
    }
    return(numeric_vec)
}

## irpp function 1 
function (rev, nb_adultes, nb_pers) 
{
    income <- 0.9334 * rev/quotient(nb_pers, nb_adultes)
    ir <- 0
    ir <- ir + (income - 12815.25 * 12) * 0.45 * (income > 12815.25 * 
        12)
    ir <- ir + (pmin(income, 12676 * 12) - 6051.42 * 12) * 0.41 * 
        (income > 6051.42 * 12)
    ir <- ir + (pmin(income, 6051.42 * 12) - 2257.17 * 12) * 
        0.3 * (income > 2257.17 * 12)
    ir <- ir + (pmin(income, 2257.17 * 12) - 817.25 * 12) * 0.14 * 
        (income > 817.25 * 12)
    ir <- quotient(nb_pers, nb_adultes) * ir
    seuil_decote <- (nb_adultes > 1) * 2585/12 + (nb_adultes <= 
        1) * 1569/12
    decote <- (ir < seuil_decote) * 0.75 * (seuil_decote - ir)
    return(pmax((ir - decote), 0))
}
